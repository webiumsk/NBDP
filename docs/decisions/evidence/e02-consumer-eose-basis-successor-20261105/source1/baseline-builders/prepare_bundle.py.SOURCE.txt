def pin(path):
    path=Path(path);raw=path.read_bytes()
    return dict(path=str(path),bytes=len(raw),sha256=hashlib.sha256(raw).hexdigest())


def put(path,value):
    path.parent.mkdir(parents=True,exist_ok=True)
    path.write_text(json.dumps(value,indent=2)+'\n');return pin(path)


def doc(p):
    return json.loads(Path(p['path'] if isinstance(p,dict) else p).read_bytes())


def methods(p):
    tree=ast.parse(p.read_bytes(),filename=str(p))
    return {c.name+'.'+m.name:ast.dump(m,include_attributes=False) for c in tree.body if isinstance(c,ast.ClassDef)
            for m in c.body if isinstance(m,ast.FunctionDef) and m.name.startswith('test_')}


def assignments(p):
    tree=ast.parse(p.read_bytes());out={}
    for n in tree.body:
        if isinstance(n,ast.Assign) and len(n.targets)==1 and isinstance(n.targets[0],ast.Name):
            try:out[n.targets[0].id]=ast.literal_eval(n.value)
            except (ValueError,TypeError):pass
    return out


def import_closure(code,runtime):
    """Stdlib AST only: complete local and virtual-program import closure."""
    paths={Path(name).stem:SOURCE/name for name in code if name.endswith('.py')}
    trees={name:ast.parse(path.read_bytes(),filename=str(path)) for name,path in paths.items()}
    test=assignments(SOURCE/'test_import_closure.py')
    trees['<isolated-program>']=ast.parse(test['ISOLATED_PROGRAM'],filename='<isolated-program>')
    exports={};graph={name:set() for name in trees};imports=[];dynamic=[]
    for name,tree in trees.items():
        symbols=set()
        for node in tree.body:
            if isinstance(node,(ast.FunctionDef,ast.ClassDef)):symbols.add(node.name)
            elif isinstance(node,ast.Assign):symbols|={t.id for t in node.targets if isinstance(t,ast.Name)}
            elif isinstance(node,(ast.Import,ast.ImportFrom)):symbols|={a.asname or a.name.split('.')[0] for a in node.names}
        exports[name]=symbols
    for name,tree in trees.items():
        for node in ast.walk(tree):
            requested=[]
            if isinstance(node,ast.Import):requested=[(a.name,None) for a in node.names]
            elif isinstance(node,ast.ImportFrom):
                assert node.level==0 and node.module is not None,('RELATIVE_IMPORT',name,node.lineno)
                requested=[(node.module,node.names)]
            for module,symbols in requested:
                top=module.split('.')[0]
                if top in paths:
                    assert module==top,('LOCAL_SUBMODULE',name,module)
                    graph[name].add(top)
                    if symbols is not None:
                        for symbol in symbols:assert symbol.name!='*' and symbol.name in exports[top],('UNKNOWN_LOCAL_SYMBOL',name,top,symbol.name)
                    target=pin(paths[top]);classification='EXACT_DECLARED_LOCAL_CODE'
                else:
                    assert top in sys.stdlib_module_names,('UNKNOWN_OR_AMBIENT_IMPORT',name,module)
                    target=None;classification='PYTHON_STDLIB_ROOT'
                imports.append(dict(source=name,line=node.lineno,module=module,classification=classification,target=target))
            if isinstance(node,ast.Call):
                assert not(isinstance(node.func,ast.Name) and node.func.id=='__import__'),('UNKNOWN_DYNAMIC_IMPORT',name,node.lineno)
                if isinstance(node.func,ast.Attribute) and node.func.attr=='import_module':
                    assert name=='whole_window' and ast.unparse(node)=='importlib.import_module(name)'
                    func=next(x for x in tree.body if isinstance(x,ast.FunctionDef) and x.name=='components')
                    previous=next(x for x in ast.parse((BASE/'whole_window.py').read_bytes()).body if isinstance(x,ast.FunctionDef) and x.name=='components')
                    assert ast.dump(func,include_attributes=False)==ast.dump(previous,include_attributes=False)
                    names=next(ast.literal_eval(x.value) for x in func.body if isinstance(x,ast.Assign) and any(isinstance(t,ast.Name) and t.id=='names' for t in x.targets))
                    assert type(names) is tuple and set(names)<=set(paths)
                    graph[name]|=set(names)
                    dynamic.append(dict(source=name,line=node.lineno,function='components',finiteModules=list(names),exactUnchangedPredecessorAST=True))
    runtime_names={Path(name).stem for name in runtime}
    def visit(roots):
        todo=list(roots);seen=set()
        while todo:
            name=todo.pop()
            if name not in seen:seen.add(name);todo.extend(graph[name]-seen)
        return seen
    assert visit(runtime_names)==runtime_names and visit(['consumer'])==runtime_names
    assert graph['capture_outputs']=={'strict','capture_contracts'}
    return dict(schema='e02-consumer-source2-complete-ast-import-closure-v1',actor=ACTOR,
        sourceFiles=[pin(p) for p in paths.values()],virtualPrograms=[dict(owner=pin(SOURCE/'test_import_closure.py'),literal='ISOLATED_PROGRAM')],
        imports=imports,graph={k:sorted(v) for k,v in graph.items()},runtimeRoots=sorted(runtime_names),
        transitiveConsumerClosure=sorted(visit(['consumer'])),finiteDynamicImports=dynamic,
        unknownAmbientRelativeOrUnboundedDynamicImports=[],targetImports=0,modelsExecuted=0,
        meaning='AST closure only, not executed imports. The separate model definitions exercise actual isolated -I -B reader loads only after a Root model release.',
        executionAuthorized=False,independentApproval=False)
