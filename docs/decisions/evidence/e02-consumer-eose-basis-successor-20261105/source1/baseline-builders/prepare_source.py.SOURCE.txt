def literals(text):
    out={}
    for n in ast.parse(text).body:
        if isinstance(n,ast.Assign) and len(n.targets)==1 and isinstance(n.targets[0],ast.Name):
            try:out[n.targets[0].id]=ast.literal_eval(n.value)
            except (ValueError,TypeError):pass
    return out


def assignments(text,changes):
    lines=text.splitlines(keepends=True);edits=[]
    for n in ast.parse(text).body:
        if isinstance(n,ast.Assign) and len(n.targets)==1 and isinstance(n.targets[0],ast.Name) and n.targets[0].id in changes:
            name=n.targets[0].id;edits.append((n.lineno-1,n.end_lineno,name+' = '+repr(changes[name])+'\n'))
    assert len(edits)==len(changes)
    for start,end,new in reversed(edits):lines[start:end]=[new]
    return ''.join(lines)
